cmake_minimum_required(VERSION 3.22)
project(mesh_vvc_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT MESHCORE_IMAGE_CODEC_SOURCES)
  set(MESHCORE_IMAGE_CODEC_SOURCES "$ENV{HOME}/.cache/meshcore-image-codecs-v1")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../vvc_sources.cmake")
set(mesh_native_dir "${CMAKE_CURRENT_LIST_DIR}/../../../android/app/src/main/cpp/image")
add_executable(mesh_vvc_smoke vvc_smoke.cpp "${mesh_native_dir}/vvc_backend.cpp")
target_include_directories(mesh_vvc_smoke PRIVATE "${mesh_native_dir}")
target_link_libraries(mesh_vvc_smoke PRIVATE vvenc vvdec)
target_compile_options(mesh_vvc_smoke PRIVATE -Wall -Wextra -fexceptions)
