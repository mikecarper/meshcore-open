cmake_minimum_required(VERSION 3.10)
project(aeic_mobile_decode LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)
add_library(aeic_entropy STATIC entropy.cpp rans/rans.cpp)
target_include_directories(aeic_entropy PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(aeic_entropy PUBLIC Threads::Threads)
add_library(aeic_recovery STATIC recovery.cpp)
target_include_directories(aeic_recovery PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

option(AEIC_ENTROPY_TEST_ONLY "Build entropy verification without ncnn" OFF)
if(NOT AEIC_ENTROPY_TEST_ONLY)
    find_package(ncnn REQUIRED CONFIG)
    add_executable(aeic_decode main.cpp png_writer.cpp mat_diagnostics.cpp tiled_attention.cpp file_hash.cpp)
    target_link_libraries(aeic_decode PRIVATE aeic_entropy aeic_recovery ncnn)
    # ncnn exports -fno-exceptions; the frontend validates files with exceptions.
    # A per-source option comes after imported target flags without changing ncnn.
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        set_source_files_properties(main.cpp png_writer.cpp mat_diagnostics.cpp mat_diagnostics_test.cpp tiled_attention.cpp tiled_attention_test.cpp recovery.cpp recovery_test.cpp file_hash.cpp file_hash_test.cpp PROPERTIES COMPILE_FLAGS "-fexceptions")
    endif()
    if(ANDROID)
        target_link_libraries(aeic_decode PRIVATE log z)
    else()
        find_package(ZLIB REQUIRED)
        target_link_libraries(aeic_decode PRIVATE ZLIB::ZLIB)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(aeic_entropy_test entropy_test.cpp)
    target_link_libraries(aeic_entropy_test PRIVATE aeic_entropy)
    add_test(NAME aeic_entropy_roundtrip COMMAND aeic_entropy_test)
    add_executable(aeic_file_hash_test file_hash_test.cpp file_hash.cpp)
    target_link_libraries(aeic_file_hash_test PRIVATE aeic_entropy)
    add_test(NAME aeic_streamed_file_hash COMMAND aeic_file_hash_test)
    add_executable(aeic_recovery_test recovery_test.cpp)
    target_link_libraries(aeic_recovery_test PRIVATE aeic_recovery)
    add_test(NAME aeic_isolated_stage_recovery COMMAND aeic_recovery_test)
    if(NOT AEIC_ENTROPY_TEST_ONLY)
        add_executable(aeic_tensor_lifetime_test tensor_lifetime_test.cpp)
        target_link_libraries(aeic_tensor_lifetime_test PRIVATE ncnn)
        add_test(NAME aeic_tensor_ownership COMMAND aeic_tensor_lifetime_test)
        add_executable(aeic_direct_allocator_test direct_allocator_test.cpp)
        target_link_libraries(aeic_direct_allocator_test PRIVATE ncnn)
        add_test(NAME aeic_direct_output_ownership COMMAND aeic_direct_allocator_test)
        add_executable(aeic_mat_diagnostics_test mat_diagnostics_test.cpp mat_diagnostics.cpp)
        target_link_libraries(aeic_mat_diagnostics_test PRIVATE aeic_entropy ncnn)
        add_test(NAME aeic_streamed_statistics COMMAND aeic_mat_diagnostics_test)
        add_executable(aeic_tiled_attention_test tiled_attention_test.cpp tiled_attention.cpp)
        set_property(TARGET aeic_tiled_attention_test PROPERTY CXX_STANDARD 17)
        target_link_libraries(aeic_tiled_attention_test PRIVATE ncnn)
        find_package(OpenMP QUIET)
        if(OpenMP_CXX_FOUND)
            target_link_libraries(aeic_tiled_attention_test PRIVATE OpenMP::OpenMP_CXX)
        endif()
        add_test(NAME aeic_exact_tiled_attention COMMAND aeic_tiled_attention_test)
        add_executable(aeic_png_test png_test.cpp png_writer.cpp)
        if(ANDROID)
            target_link_libraries(aeic_png_test PRIVATE z)
        else()
            target_link_libraries(aeic_png_test PRIVATE ZLIB::ZLIB)
        endif()
        add_test(NAME aeic_png_roundtrip COMMAND aeic_png_test)
    endif()
endif()
